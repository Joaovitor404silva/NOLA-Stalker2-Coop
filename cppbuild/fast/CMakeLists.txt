cmake_minimum_required(VERSION 3.27)
cmake_policy(SET CMP0141 NEW)
cmake_policy(SET CMP0099 NEW)
project(UE4SSNativeModsFast)
if(NOT NOLA_UE4SS_LIBDIR)
    message(FATAL_ERROR "NOLA_UE4SS_LIBDIR (the UE4SS cache lib folder) is required")
endif()
if(CMAKE_CXX_COMPILER_LAUNCHER)
    file(WRITE "${CMAKE_BINARY_DIR}/si_probe/si.h" "")
    file(WRITE "${CMAKE_BINARY_DIR}/si_probe/si.cpp" "#include \"si.h\"\n")
    execute_process(COMMAND "${CMAKE_CXX_COMPILER_LAUNCHER}" "${CMAKE_CXX_COMPILER}" /nologo /showIncludes /c si.cpp
        WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/si_probe" OUTPUT_VARIABLE si_out ERROR_VARIABLE si_out ENCODING UTF8)
    if(si_out MATCHES "\n([^\n]*[^ \n]) +[A-Za-z]:[\\/][^\n]*si\\.h")
        set(CMAKE_CL_SHOWINCLUDES_PREFIX "${CMAKE_MATCH_1} ")
        message(STATUS "showIncludes prefix through the launcher: '${CMAKE_CL_SHOWINCLUDES_PREFIX}'")
    else()
        message(WARNING "showIncludes prefix through the launcher not found; header edits will not rebuild")
    endif()
endif()
add_compile_definitions(IMGUI_DEFINE_MATH_OPERATORS)
add_subdirectory(D:/modding/cpp/RE-UE4SS ${CMAKE_BINARY_DIR}/RE-UE4SS EXCLUDE_FROM_ALL)
get_filename_component(S2_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../src" ABSOLUTE)
add_subdirectory(${S2_SRC} ${CMAKE_BINARY_DIR}/NolaNative)

get_target_property(ue4ss_opts UE4SS INTERFACE_COMPILE_OPTIONS)
string(REPLACE ";/Zi;" ";" ue4ss_opts ";${ue4ss_opts};")
string(REPLACE ";/MP;" ";" ue4ss_opts "${ue4ss_opts}")
string(REGEX REPLACE "^;|;$" "" ue4ss_opts "${ue4ss_opts}")
set_property(TARGET UE4SS PROPERTY INTERFACE_COMPILE_OPTIONS "${ue4ss_opts}")

set(nola_libs UE4SS Unreal ASMHelper SinglePassSigScanner LuaMadeSimple LuaRaw IniParser JSON ParserBase Input DynamicOutput File Helpers patternsleuth_bind
    fmt ImGui PolyHook_2 Zydis Zycore asmtk asmjit glfw3 glad)
set(nola_link "$<COMPILE_ONLY:UE4SS>" ws2_32)
foreach(l ${nola_libs})
    list(APPEND nola_link "${NOLA_UE4SS_LIBDIR}/${l}.lib")
endforeach()
list(APPEND nola_link kernel32 d3dcompiler dxgi d3d11 opengl32 dbghelp psapi ntdll userenv user32 gdi32 winspool shell32 ole32 oleaut32 uuid comdlg32 advapi32)
set_property(TARGET NolaNative PROPERTY LINK_LIBRARIES "${nola_link}")
set_property(TARGET NolaNative PROPERTY INTERFACE_LINK_LIBRARIES "")
set_property(TARGET NolaNative PROPERTY MSVC_DEBUG_INFORMATION_FORMAT Embedded)
target_link_options(NolaNative PRIVATE ${NOLA_FAST_LINK})
