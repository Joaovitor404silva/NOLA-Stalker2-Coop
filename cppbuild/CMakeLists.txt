cmake_minimum_required(VERSION 3.18)
project(UE4SSNativeMods)
add_compile_definitions(IMGUI_DEFINE_MATH_OPERATORS)
set(UE4SS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../RE-UE4SS" CACHE PATH "RE-UE4SS checkout (tested upstream revision 5e62799)")
if(NOT EXISTS "${UE4SS_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Set -DUE4SS_ROOT to a RE-UE4SS checkout with submodules initialized")
endif()
add_subdirectory("${UE4SS_ROOT}" ${CMAKE_BINARY_DIR}/RE-UE4SS)
get_filename_component(S2_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../src" ABSOLUTE)
add_subdirectory(${S2_SRC} ${CMAKE_BINARY_DIR}/NolaNative)
